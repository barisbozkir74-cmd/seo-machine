```typescript
'use server'

import { revalidatePath } from 'next/cache'
import { createClient } from '@/lib/supabase/server'

export type RecoveryTaskActionResult =
  | { success: true }
  | { success: false; error: string }

/**
 * D-11, D-13, UI-SPEC §"Güncelle" / "Görmezden Gel" Click Flow.
 *
 * Updates a recovery task's status — invoked from RecoveryTaskTable.
 * - 'in_progress': user clicked Güncelle (also navigates client-side to editor)
 * - 'dismissed': user clicked Görmezden Gel (row removed from default view)
 *
 * Auth: session-bound. Ownership: project must belong to the calling user.
 * Race: re-checks ownership server-side even though RLS enforces it (defence-in-depth).
 *
 * 'open' and 'resolved' are NOT valid targets for this action — open is the seed
 * status set by detect route, resolved is set automatically by publishToWordPress.
 */
export async function updateRecoveryTaskStatus(
  taskId: string,
  projectId: string,
  status: 'in_progress' | 'dismissed',
): Promise<RecoveryTaskActionResult> {
  // Input validation — these should already be UUIDs from the typed prop chain,
  // but a defence-in-depth shape check costs nothing.
  if (!taskId || typeof taskId !== 'string') {
    return { success: false, error: 'Görev kimliği geçersiz.' }
  }
  if (!projectId || typeof projectId !== 'string') {
    return { success: false, error: 'Proje kimliği geçersiz.' }
  }
  if (status !== 'in_progress' && status !== 'dismissed') {
    return { success: false, error: 'Geçersiz durum.' }
  }

  const supabase = await createClient()

  // 1. Authenticate
  const {
    data: { user },
  } = await supabase.auth.getUser()
  if (!user) {
    return { success: false, error: 'Oturum bulunamadı.' }
  }

  // 2. Ownership: verify the project belongs to this user
  const { data: project } = await supabase
    .from('projects')
    .select('id')
    .eq('id', projectId)
    .eq('user_id', user.id)
    .single()
  if (!project) {
    return { success: false, error: 'Proje bulunamadı.' }
  }

  // 3. Mutate. RLS UPDATE policy (Plan 16-01) re-checks ownership in addition.
  //    The .eq('project_id', projectId) ensures the task belongs to the verified project.
  const { error, data } = await supabase
    .from('recovery_tasks')
    .update({
      status,
      updated_at: new Date().toISOString(),
    })
    .eq('id', taskId)
    .eq('project_id', projectId)
    // Only allow transitions from 'open' or 'in_progress' — never overwrite a resolved task
    .in('status', ['open', 'in_progress'])
    .select('id')

  if (error) {
    console.error('[updateRecoveryTaskStatus]', error.message)
    return { success: false, error: 'Görev güncellenemedi.' }
  }

  if (!data || data.length === 0) {
    // Task not found, not owned, or already in a terminal state (resolved)
    return { success: false, error: 'Görev güncellenemedi (bulunamadı veya zaten çözüldü).' }
  }

  revalidatePath(`/projeler/${projectId}/izleme`)
  return { success: true }
}
```

**Critical rules:**
1. The function signature is IDENTICAL to the stub from 16-03 — RecoveryTaskTable's import does not need to change.
2. Status `'open'` and `'resolved'` are explicitly REJECTED at the input-validation step. Only the user-facing transitions (in_progress, dismissed) are allowed via this action. open is set by detect route. resolved is set by publishToWordPress.
3. The `.in('status', ['open', 'in_progress'])` filter on the UPDATE prevents overwriting a row that has already become 'resolved' (race with publishToWordPress).
4. `.select('id')` after the UPDATE allows us to detect "no rows matched" — a missing task or a stale state. We surface this to the user as a recoverable error rather than a silent failure.
5. Error message strings are Turkish and match project convention ("Oturum bulunamadı.", "Proje bulunamadı.").
6. Use `createClient()` from `@/lib/supabase/server` — NOT service role. RLS gates apply, and ownership re-check is defence-in-depth.
7. `revalidatePath` is necessary so SSR refresh in RecoveryTaskTable's `router.refresh()` actually re-fetches data.

import Link from 'next/link'

import { getClusterMetrics, getPageMetrics, type MonitoringPeriod } from '@/lib/monitoring/aggregation'
import { getRecoveryTasks } from '@/lib/monitoring/recovery-tasks'
import { ProjectNav } from '../ProjectNav'
import { PeriodTabBar } from './period-tab-bar'
import { ContentTabBar, type ContentTab } from './content-tab-bar'
import { ClusterSummaryTable } from './cluster-summary-table'
import { PageMetricsTable } from './page-metrics-table'
import { RecoveryTaskTable } from './recovery-task-table'

export default async function IzlemePage({
  params,
  searchParams,
}: {
  params: Promise<{ id: string }>
  searchParams: Promise<{ period?: string; tab?: string }>
}) {
  const { id } = await params
  const { period: periodRaw, tab: tabRaw } = await searchParams

  const period: MonitoringPeriod =
    periodRaw === '7' ? 7 : periodRaw === '90' ? 90 : 28

  // UI-SPEC §URL State Machine: default tab=clusters when missing/invalid (was the original
  // single-section default). Wave 16-04 introduces the tab param — backward compat preserved.
  const activeTab: ContentTab =
    tabRaw === 'pages'
      ? 'pages'
      : tabRaw === 'recovery'
      ? 'recovery'
      : 'clusters'

  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) notFound()

  const { data: project } = await supabase
    .from('projects')
    .select('id, name, domain, gsc_property_url')
    .eq('id', id)
    .eq('user_id', user.id)
    .single()
  if (!project) notFound()

  const gscConnected =
    project.gsc_property_url !== null && project.gsc_property_url !== ''

  // Parallel fetch: only when GSC connected (recovery tasks may exist independently —
  // imported_page source does not require GSC, but page_package source does — we still
  // gate on gscConnected because the izleme page itself is GSC-centered. Recovery tab
  // shows empty state if list is [].)
  const [clusters, pages, recoveryTasks] = gscConnected
    ? await Promise.all([
        getClusterMetrics(supabase, id, period),
        getPageMetrics(supabase, id, period),
        getRecoveryTasks(supabase, id),
      ])
    : [[], [], []]

  return (
    <div className="flex flex-col h-screen">
      <div className="p-8 pb-4">
        <Link
          href={`/projeler/${id}`}
          className="text-sm text-muted-foreground hover:text-foreground flex items-center gap-1 mb-4"
        >
          ← {project.name}
        </Link>
        <h1 className="text-xl font-semibold">İzleme</h1>
      </div>

      <div className="flex flex-1 min-h-0">
        <div className="w-64 shrink-0 border-r border-border overflow-y-auto p-4">
          <ProjectNav projectId={id} activePath={`/projeler/${id}/izleme`} />
        </div>

        <div className="flex-1 min-w-0 overflow-y-auto p-8">
          {!gscConnected ? (
            <div className="rounded-lg border border-border bg-card p-8 text-center">
              <p className="text-base font-semibold">GSC verisi bulunamadı</p>
              <p className="text-sm text-muted-foreground mt-2">
                Bu proje için henüz GSC metriği senkronize edilmedi. Proje Bilgileri sayfasından GSC bağlantısını kontrol edin.
              </p>
              <Link
                href={`/projeler/${id}`}
                className="inline-flex items-center mt-4 text-sm text-foreground hover:underline"
              >
                Proje Bilgileri →
              </Link>
            </div>
          ) : (
            <div className="space-y-8">
              <PeriodTabBar projectId={id} active={period} />
              <ContentTabBar projectId={id} active={activeTab} period={period} />

              {activeTab === 'clusters' && (
                <section>
                  <h2 className="text-base font-semibold mb-4">Cluster Performansı</h2>
                  <ClusterSummaryTable clusters={clusters} />
                </section>
              )}

              {activeTab === 'pages' && (
                <section>
                  <h2 className="text-base font-semibold mb-4">Sayfa Performansı</h2>

                </section>

                  <h2 className="text-base font-semibold mb-4">Recovery</h2>
                  <RecoveryTaskTable tasks={recoveryTasks} projectId={id} />
                </section>
              )}
            </div>
          )}
        </div>
      </div>
    </div>
  )
}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser → /projeler/[id]/izleme | SSR page; supabase auth from cookies. notFound() on missing user or non-owned project |
| browser → updateRecoveryTaskStatus | Server action; auth from cookies. Explicit ownership re-check + RLS UPDATE policy |
| URL ?tab= param | User-controlled string; whitelisted to ('clusters' \| 'pages' \| 'recovery') |
| URL ?period= param | User-controlled string; whitelisted to ('7' \| '28' \| '90') |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-16-04-01 | Spoofing | unauth user accesses /izleme | mitigate | `supabase.auth.getUser()` + `notFound()` if null. Middleware also redirects unauth users to /login. |
| T-16-04-02 | Information Disclosure | user reads another user's project's izleme | mitigate | Project SELECT filtered by `.eq('user_id', user.id)`; notFound() if no row. |
| T-16-04-03 | Information Disclosure | user reads another user's recovery_tasks via getRecoveryTasks | mitigate | RLS SELECT policy on recovery_tasks (Plan 16-01) + `getRecoveryTasks` filters by `project_id` which is verified owned. |
| T-16-04-04 | Tampering | malformed ?tab= renders unexpected section | mitigate | Whitelist parser falls back to 'clusters' for any unknown value. |
| T-16-04-05 | Tampering | updateRecoveryTaskStatus called with wrong projectId | mitigate | Server action re-verifies project ownership + filters task UPDATE by both `id` and `project_id`. RLS UPDATE policy is defence-in-depth. |
| T-16-04-06 | Tampering | replay attack on dismiss action | mitigate | UPDATE filtered by `.in('status', ['open', 'in_progress'])` — replay on a resolved task is rejected. |
| T-16-04-07 | Tampering | bypass auth via direct PostgREST POST | mitigate | RLS UPDATE policy uses `EXISTS (… projects p WHERE p.user_id = auth.uid())`. Anon key without session cookie has no auth.uid() — UPDATE fails. |
| T-16-04-08 | Repudiation | server action error logging | accept | console.error logs error.message only — no PII. Adequate for MVP audit trail. |
| T-16-04-09 | Information Disclosure | error message leaking DB internals | mitigate | Errors are mapped to fixed Turkish strings; raw error.message stays in console.error, not in the user response. |
| T-16-04-10 | DoS | concurrent updateRecoveryTaskStatus calls | accept | Single-row UPDATE on indexed columns. No realistic DoS surface at MVP scale. |
</threat_model>
