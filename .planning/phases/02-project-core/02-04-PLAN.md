- Aktif: `<Record01Icon size={14} className="text-blue-400 shrink-0" />` (circle filled)

**Veri sorgusu:**
```typescript
const supabase = await createClient()

const { data: project } = await supabase
  .from('projects')
  .select('id, name, domain, sector, target_country, target_language, business_model, site_type, brand_tone, notes, created_at')
  .eq('id', id)
  .single()

if (!project) notFound()

const { data: stages } = await supabase
  .from('stages')
  .select('id, stage_name, status, started_at, completed_at')
  .eq('project_id', id)
  .order('created_at', { ascending: true })
```
